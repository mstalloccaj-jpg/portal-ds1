-- ============================================================ -- PORTAL DS-1 — GM Robótica — Schema v2.0 -- NOVEDADES: Empresas, Roles por empresa, Sistema de invitaciones -- Ejecutar en Supabase → SQL Editor (reemplaza el schema anterior) -- ============================================================ -- ── Limpiar tablas anteriores si existen ── DROP TABLE IF EXISTS audit_logs CASCADE; DROP TABLE IF EXISTS report_assignments CASCADE; DROP TABLE IF EXISTS reports CASCADE; DROP TABLE IF EXISTS profiles CASCADE; DROP TABLE IF EXISTS invitations CASCADE; DROP TABLE IF EXISTS companies CASCADE; DROP FUNCTION IF EXISTS handle_new_user CASCADE; DROP FUNCTION IF EXISTS is_admin CASCADE; DROP FUNCTION IF EXISTS is_company_admin CASCADE; DROP FUNCTION IF EXISTS update_updated_at CASCADE; -- ============================================================ -- 1. EMPRESAS -- ============================================================ CREATE TABLE companies ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name TEXT NOT NULL, rut_cuit TEXT, address TEXT, city TEXT, country TEXT DEFAULT 'Argentina', phone TEXT, email TEXT, logo_url TEXT, active BOOLEAN NOT NULL DEFAULT TRUE, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- Empresa inicial: GM Robótica (la inspectora) INSERT INTO companies (id, name, city, email, active) VALUES ('00000000-0000-0000-0000-000000000001', 'GM Robótica', 'Mendoza', 'info@gmrobotica.ar', TRUE); -- ============================================================ -- 2. PERFILES DE USUARIO -- Roles del sistema: -- super_admin → GM Robótica, acceso total -- company_admin → admin de una empresa cliente -- inspector → puede crear y subir reportes -- supervisor → aprueba reportes, ve todo de su empresa -- viewer → solo lee/descarga reportes asignados -- ============================================================ CREATE TABLE profiles ( id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE, company_id UUID REFERENCES companies(id) ON DELETE SET NULL, full_name TEXT NOT NULL DEFAULT '', position TEXT, -- cargo / posición phone TEXT, role TEXT NOT NULL DEFAULT 'viewer' CHECK (role IN ('super_admin','company_admin','inspector','supervisor','viewer')), active BOOLEAN NOT NULL DEFAULT TRUE, must_change_password BOOLEAN NOT NULL DEFAULT FALSE, -- fuerza cambio de contraseña last_login TIMESTAMPTZ, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- ============================================================ -- 3. INVITACIONES (sistema seguro de incorporación) -- El admin crea la invitación → el sistema manda el email -- El usuario usa el token para activar su cuenta y poner contraseña -- ============================================================ CREATE TABLE invitations ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), email TEXT NOT NULL, full_name TEXT NOT NULL, company_id UUID REFERENCES companies(id) ON DELETE CASCADE, role TEXT NOT NULL DEFAULT 'viewer' CHECK (role IN ('company_admin','inspector','supervisor','viewer')), token TEXT NOT NULL UNIQUE DEFAULT encode(gen_random_bytes(32), 'hex'), invited_by UUID REFERENCES profiles(id) ON DELETE SET NULL, accepted BOOLEAN NOT NULL DEFAULT FALSE, expires_at TIMESTAMPTZ NOT NULL DEFAULT (NOW() + INTERVAL '7 days'), created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- ============================================================ -- 4. REPORTES DE INSPECCIÓN -- ============================================================ CREATE TABLE reports ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), -- Identificación report_no TEXT NOT NULL, date DATE, revision TEXT DEFAULT '0', -- Componente component TEXT, type TEXT, serial_no TEXT, heat_no TEXT, manufacturer TEXT, material TEXT, grade TEXT, nom_od TEXT, nom_id TEXT, comp_length TEXT, weight TEXT, makeup_torque TEXT, conn_top TEXT, conn_bot TEXT, -- Contexto operativo company_id UUID REFERENCES companies(id) ON DELETE SET NULL, -- empresa CLIENTE client_company_name TEXT, rig TEXT, well TEXT, location TEXT, -- Inspector inspector TEXT, cert_no TEXT, category TEXT DEFAULT '3', -- Resultado result TEXT DEFAULT 'APROBADO' CHECK (result IN ('APROBADO','RECHAZADO','CONDICIONAL')), comments TEXT, -- Datos estructurados DS-1 visual_inspection JSONB DEFAULT '{}', measurements JSONB DEFAULT '{}', images JSONB DEFAULT '[]', -- URLs de imágenes adjuntas raw_data JSONB DEFAULT '{}', -- Control published BOOLEAN NOT NULL DEFAULT FALSE, uploaded_by UUID REFERENCES profiles(id) ON DELETE SET NULL, approved_by UUID REFERENCES profiles(id) ON DELETE SET NULL, approved_at TIMESTAMPTZ, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- ============================================================ -- 5. ASIGNACIONES (reportes visibles para usuarios específicos) -- ============================================================ CREATE TABLE report_assignments ( report_id UUID NOT NULL REFERENCES reports(id) ON DELETE CASCADE, profile_id UUID NOT NULL REFERENCES profiles(id) ON DELETE CASCADE, assigned_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), PRIMARY KEY (report_id, profile_id) ); -- ============================================================ -- 6. AUDIT LOG (inmutable — solo INSERT) -- ============================================================ CREATE TABLE audit_logs ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), event TEXT NOT NULL, user_id UUID REFERENCES profiles(id) ON DELETE SET NULL, user_name TEXT, user_company TEXT, user_role TEXT, report_id UUID REFERENCES reports(id) ON DELETE SET NULL, report_no TEXT, company_id UUID REFERENCES companies(id) ON DELETE SET NULL, ip_address TEXT, user_agent TEXT, metadata JSONB DEFAULT '{}', created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- Solo se puede insertar, nunca modificar ni borrar CREATE RULE audit_log_no_update AS ON UPDATE TO audit_logs DO INSTEAD NOTHING; CREATE RULE audit_log_no_delete AS ON DELETE TO audit_logs DO INSTEAD NOTHING; -- ============================================================ -- 7. TRIGGERS -- ============================================================ CREATE OR REPLACE FUNCTION update_updated_at() RETURNS TRIGGER AS $$ BEGIN NEW.updated_at = NOW(); RETURN NEW; END; $$ LANGUAGE plpgsql; CREATE TRIGGER trg_companies_updated_at BEFORE UPDATE ON companies FOR EACH ROW EXECUTE FUNCTION update_updated_at(); CREATE TRIGGER trg_profiles_updated_at BEFORE UPDATE ON profiles FOR EACH ROW EXECUTE FUNCTION update_updated_at(); CREATE TRIGGER trg_reports_updated_at BEFORE UPDATE ON reports FOR EACH ROW EXECUTE FUNCTION update_updated_at(); -- Trigger: crear perfil automáticamente al registrarse CREATE OR REPLACE FUNCTION handle_new_user() RETURNS TRIGGER AS $$ DECLARE v_invite invitations%ROWTYPE; v_company_id UUID; v_role TEXT := 'viewer'; v_name TEXT; BEGIN -- Buscar invitación válida por email SELECT * INTO v_invite FROM invitations WHERE email = NEW.email AND accepted = FALSE AND expires_at > NOW() ORDER BY created_at DESC LIMIT 1; IF FOUND THEN v_company_id := v_invite.company_id; v_role := v_invite.role; v_name := v_invite.full_name; -- Marcar invitación como aceptada UPDATE invitations SET accepted = TRUE WHERE id = v_invite.id; ELSE -- Sin invitación: viewer sin empresa v_company_id := NULL; v_role := 'viewer'; v_name := COALESCE(NEW.raw_user_meta_data->>'full_name', split_part(NEW.email,'@',1)); END IF; INSERT INTO profiles (id, company_id, full_name, role, must_change_password) VALUES (NEW.id, v_company_id, v_name, v_role, TRUE) ON CONFLICT (id) DO NOTHING; RETURN NEW; EXCEPTION WHEN OTHERS THEN RETURN NEW; END; $$ LANGUAGE plpgsql SECURITY DEFINER; CREATE OR REPLACE TRIGGER trg_on_auth_user_created AFTER INSERT ON auth.users FOR EACH ROW EXECUTE FUNCTION handle_new_user(); -- ============================================================ -- 8. FUNCIONES DE SEGURIDAD (helper para RLS) -- ============================================================ CREATE OR REPLACE FUNCTION current_profile() RETURNS profiles AS $$ SELECT * FROM profiles WHERE id = auth.uid(); $$ LANGUAGE sql SECURITY DEFINER STABLE; CREATE OR REPLACE FUNCTION is_super_admin() RETURNS BOOLEAN AS $$ SELECT EXISTS (SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'super_admin'); $$ LANGUAGE sql SECURITY DEFINER STABLE; CREATE OR REPLACE FUNCTION is_admin_or_super() RETURNS BOOLEAN AS $$ SELECT EXISTS (SELECT 1 FROM profiles WHERE id = auth.uid() AND role IN ('super_admin','company_admin')); $$ LANGUAGE sql SECURITY DEFINER STABLE; CREATE OR REPLACE FUNCTION my_company_id() RETURNS UUID AS $$ SELECT company_id FROM profiles WHERE id = auth.uid(); $$ LANGUAGE sql SECURITY DEFINER STABLE; -- ============================================================ -- 9. ROW LEVEL SECURITY (RLS) -- ============================================================ ALTER TABLE companies ENABLE ROW LEVEL SECURITY; ALTER TABLE profiles ENABLE ROW LEVEL SECURITY; ALTER TABLE invitations ENABLE ROW LEVEL SECURITY; ALTER TABLE reports ENABLE ROW LEVEL SECURITY; ALTER TABLE report_assignments ENABLE ROW LEVEL SECURITY; ALTER TABLE audit_logs ENABLE ROW LEVEL SECURITY; -- COMPANIES CREATE POLICY "companies_super_admin" ON companies FOR ALL USING (is_super_admin()); CREATE POLICY "companies_read_own" ON companies FOR SELECT USING (id = my_company_id() OR is_super_admin()); -- PROFILES CREATE POLICY "profiles_super_admin" ON profiles FOR ALL USING (is_super_admin()); CREATE POLICY "profiles_read_same_company" ON profiles FOR SELECT USING (company_id = my_company_id() OR id = auth.uid() OR is_super_admin()); CREATE POLICY "profiles_update_own" ON profiles FOR UPDATE USING (id = auth.uid()); CREATE POLICY "profiles_company_admin_manage" ON profiles FOR ALL USING (company_id = my_company_id() AND is_admin_or_super()); -- INVITATIONS CREATE POLICY "invitations_super_admin" ON invitations FOR ALL USING (is_super_admin()); CREATE POLICY "invitations_company_admin" ON invitations FOR ALL USING (company_id = my_company_id() AND is_admin_or_super()); CREATE POLICY "invitations_read_own" ON invitations FOR SELECT USING (invited_by = auth.uid() OR is_super_admin()); -- REPORTS CREATE POLICY "reports_super_admin" ON reports FOR ALL USING (is_super_admin()); CREATE POLICY "reports_company_all" ON reports FOR ALL USING (company_id = my_company_id() AND is_admin_or_super()); CREATE POLICY "reports_inspector_create" ON reports FOR INSERT WITH CHECK (EXISTS (SELECT 1 FROM profiles WHERE id = auth.uid() AND role IN ('inspector','company_admin','super_admin'))); CREATE POLICY "reports_assigned_read" ON reports FOR SELECT USING ( is_super_admin() OR company_id = my_company_id() OR EXISTS (SELECT 1 FROM report_assignments ra WHERE ra.report_id = reports.id AND ra.profile_id = auth.uid()) ); -- REPORT ASSIGNMENTS CREATE POLICY "ra_super_admin" ON report_assignments FOR ALL USING (is_super_admin()); CREATE POLICY "ra_admin_manage" ON report_assignments FOR ALL USING (is_admin_or_super()); CREATE POLICY "ra_read_own" ON report_assignments FOR SELECT USING (profile_id = auth.uid() OR is_admin_or_super()); -- AUDIT LOGS CREATE POLICY "audit_super_admin" ON audit_logs FOR ALL USING (is_super_admin()); CREATE POLICY "audit_company_read" ON audit_logs FOR SELECT USING (company_id = my_company_id() AND is_admin_or_super()); CREATE POLICY "audit_insert_own" ON audit_logs FOR INSERT WITH CHECK (user_id = auth.uid() OR is_super_admin()); STORAGE BUCKET para imágenes de reportes -- ============================================================ INSERT INTO storage.buckets (id, name, public, file_size_limit, allowed_mime_types) VALUES ( 'report-images', 'report-images', FALSE, 10485760, -- 10MB max por imagen ARRAY['image/jpeg','image/png','image/webp','image/heic'] ) ON CONFLICT (id) DO NOTHING; -- Storage RLS CREATE POLICY "report_images_upload" ON storage.objects FOR INSERT WITH CHECK (bucket_id = 'report-images' AND auth.role() = 'authenticated'); CREATE POLICY "report_images_read" ON storage.objects FOR SELECT USING (bucket_id = 'report-images' AND auth.role() = 'authenticated'); CREATE POLICY "report_images_delete" ON storage.objects FOR DELETE USING (bucket_id = 'report-images' AND (auth.uid())::text = (storage.foldername(name))[1]); -- ============================================================ -- 11. ÍNDICES -- ============================================================ CREATE INDEX IF NOT EXISTS idx_profiles_company ON profiles(company_id); CREATE INDEX IF NOT EXISTS idx_profiles_role ON profiles(role); CREATE INDEX IF NOT EXISTS idx_reports_company ON reports(company_id); CREATE INDEX IF NOT EXISTS idx_reports_result ON reports(result); CREATE INDEX IF NOT EXISTS idx_reports_date ON reports(date); CREATE INDEX IF NOT EXISTS idx_reports_published ON reports(published); CREATE INDEX IF NOT EXISTS idx_ra_profile ON report_assignments(profile_id); CREATE INDEX IF NOT EXISTS idx_invitations_email ON invitations(email); CREATE INDEX IF NOT EXISTS idx_invitations_token ON invitations(token); CREATE INDEX IF NOT EXISTS idx_audit_company ON audit_logs(company_id); CREATE INDEX IF NOT EXISTS idx_audit_created ON audit_logs(created_at DESC); -- ============================================================ -- 12.
